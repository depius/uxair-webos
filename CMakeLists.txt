cmake_minimum_required(VERSION 3.16)
project(uxair_webos LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED sdl2)
pkg_check_modules(TTF REQUIRED SDL2_ttf)
pkg_check_modules(LS2 REQUIRED luna-service2 glib-2.0)   # screen saver, foreground (src/system_bus.cpp)
# The video/audio player, one per build: ndl, LG's NDL direct media API, for
# webOS 5 and later (webOS 10 has nothing else); starfish, the Starfish media
# pipeline + ACB, for webOS 3-4. (SDL2_image, for cover art, is loaded at run
# time: an older TV without it shows no cover.)
set(UXAIR_MEDIA "ndl" CACHE STRING "Player: ndl (webOS 5+) or starfish (webOS 3-4)")
set_property(CACHE UXAIR_MEDIA PROPERTY STRINGS ndl starfish)
set(NDL_INCLUDE_DIR "${CMAKE_SYSROOT}/usr/include/libndl-media")
if(UXAIR_MEDIA STREQUAL "ndl")
    set(MEDIA_SOURCES src/media_ndl.cpp src/ndl_backend.c)
    set(MEDIA_DEFINITIONS UXAIR_MEDIA_NDL)
    set(MEDIA_INCLUDE_DIRS "${NDL_INCLUDE_DIR}")
    set(MEDIA_LIBRARY_DIRS "")
    set(MEDIA_LIBRARIES NDL_directmedia)
    set_source_files_properties(src/ndl_backend.c PROPERTIES COMPILE_OPTIONS "-Wno-attributes")
elseif(UXAIR_MEDIA STREQUAL "starfish")
    pkg_check_modules(SMP REQUIRED libplayerAPIs)
    pkg_check_modules(ACB REQUIRED AcbAPI)
    set(MEDIA_SOURCES src/media_starfish.cpp)
    set(MEDIA_DEFINITIONS UXAIR_MEDIA_STARFISH)
    set(MEDIA_INCLUDE_DIRS ${SMP_INCLUDE_DIRS} ${ACB_INCLUDE_DIRS})
    set(MEDIA_LIBRARY_DIRS ${SMP_LIBRARY_DIRS} ${ACB_LIBRARY_DIRS})
    set(MEDIA_LIBRARIES ${SMP_LIBRARIES} ${ACB_LIBRARIES})
else()
    message(FATAL_ERROR "UXAIR_MEDIA must be ndl or starfish, not ${UXAIR_MEDIA}")
endif()
add_library(lg_media STATIC ${MEDIA_SOURCES} src/shell.cpp src/input.cpp src/settings.cpp src/ui_navigation.cpp src/usage_stats.cpp)
target_compile_definitions(lg_media PUBLIC _GLIBCXX_USE_CXX11_ABI=0 ${MEDIA_DEFINITIONS})
target_include_directories(lg_media PUBLIC src ${SDL_INCLUDE_DIRS} ${TTF_INCLUDE_DIRS} PRIVATE ${MEDIA_INCLUDE_DIRS})
target_link_directories(lg_media PUBLIC ${SDL_LIBRARY_DIRS} ${MEDIA_LIBRARY_DIRS})
target_link_libraries(lg_media PUBLIC ${SDL_LIBRARIES} ${TTF_LIBRARIES} ${MEDIA_LIBRARIES} pthread dl)
target_compile_options(lg_media PRIVATE -Wall -Wextra)
add_executable(media-probe src/probe.cpp)
target_link_libraries(media-probe PRIVATE lg_media)
set_target_properties(media-probe PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
if(UXAIR_MEDIA STREQUAL "ndl")
    # plays a test stream through NDL on the TV (tools/package-ndl-probe.py)
    add_executable(ndl-probe src/ndl_probe.c)
    target_include_directories(ndl-probe PRIVATE "${NDL_INCLUDE_DIR}" ${SDL_INCLUDE_DIRS})
    target_link_directories(ndl-probe PRIVATE ${SDL_LIBRARY_DIRS})
    target_link_libraries(ndl-probe PRIVATE NDL_directmedia ${SDL_LIBRARIES} pthread dl)
    target_compile_options(ndl-probe PRIVATE -Wall -Wextra -Wno-attributes)
    set_target_properties(ndl-probe PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
option(BUILD_HEVC_PROBE "Build isolated manually launched HEVC hardware probe" OFF)
if(BUILD_HEVC_PROBE)
    pkg_check_modules(SMP REQUIRED libplayerAPIs)
    pkg_check_modules(ACB REQUIRED AcbAPI)
    add_executable(hevc-probe src/hevc_probe.cpp src/hevc_sample.cpp src/hevc_launch.cpp src/hevc_gate.cpp)
    target_compile_definitions(hevc-probe PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
    target_include_directories(hevc-probe PRIVATE src ${SDL_INCLUDE_DIRS} ${SMP_INCLUDE_DIRS} ${ACB_INCLUDE_DIRS})
    target_link_directories(hevc-probe PRIVATE ${SDL_LIBRARY_DIRS} ${SMP_LIBRARY_DIRS} ${ACB_LIBRARY_DIRS})
    target_link_libraries(hevc-probe PRIVATE ${SDL_LIBRARIES} ${SMP_LIBRARIES} ${ACB_LIBRARIES} pthread)
    target_compile_options(hevc-probe PRIVATE -Wall -Wextra)
    set_target_properties(hevc-probe PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
option(BUILD_RECEIVER "Build the AirPlay receiver (requires tools/build-deps.sh)" OFF)
option(BUILD_DEVELOPMENT_PACKAGE "Use a separate development app and AirPlay identity" OFF)
if(BUILD_DEVELOPMENT_PACKAGE)
    target_compile_definitions(lg_media PUBLIC UXAIR_DEV_BUILD)
endif()
if(BUILD_RECEIVER)
    set(DEPS "${CMAKE_SOURCE_DIR}/.cache/deps" CACHE PATH "Private ARM dependencies")
    set(UX "${CMAKE_SOURCE_DIR}/vendor/uxplay")
    add_subdirectory(${UX}/lib/llhttp)
    add_subdirectory(${UX}/lib/playfair)
    add_subdirectory(${UX}/lib/mdnsd)
    file(GLOB CORE CONFIGURE_DEPENDS "${UX}/lib/*.c")
    if(BUILD_DEVELOPMENT_PACKAGE)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        execute_process(
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/instrument-uxplay.py" "${UX}/lib" "${CMAKE_BINARY_DIR}/generated"
            RESULT_VARIABLE PROBE_RESULT
        )
        if(NOT PROBE_RESULT EQUAL 0)
            message(FATAL_ERROR "Pinned UxPlay direct-request instrumentation failed")
        endif()
        foreach(PROBE_FILE raop.c http_request.c httpd.c raop_rtp_mirror.c)
            list(REMOVE_ITEM CORE "${UX}/lib/${PROBE_FILE}")
            list(APPEND CORE "${CMAKE_BINARY_DIR}/generated/${PROBE_FILE}")
        endforeach()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
            "${CMAKE_SOURCE_DIR}/tools/instrument-uxplay.py"
            "${UX}/lib/raop.c" "${UX}/lib/http_request.c" "${UX}/lib/httpd.c" "${UX}/lib/raop_rtp_mirror.c")
    endif()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(LIFECYCLE_INPUT "${UX}/lib/raop.c")
    if(BUILD_DEVELOPMENT_PACKAGE)
        set(LIFECYCLE_INPUT "${CMAKE_BINARY_DIR}/generated/raop.c")
    endif()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/instrument-lifecycle.py"
        "${UX}/lib" "${LIFECYCLE_INPUT}" "${CMAKE_BINARY_DIR}/generated"
        RESULT_VARIABLE LIFECYCLE_RESULT)
    if(NOT LIFECYCLE_RESULT EQUAL 0)
        message(FATAL_ERROR "Pinned UxPlay lifecycle instrumentation failed")
    endif()
    list(REMOVE_ITEM CORE "${UX}/lib/raop.c" "${CMAKE_BINARY_DIR}/generated/raop.c")
    list(APPEND CORE "${CMAKE_BINARY_DIR}/generated/lifecycle-raop.c")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_SOURCE_DIR}/tools/instrument-lifecycle.py" "${UX}/lib/raop_handlers.h" "${UX}/lib/raop.c")
    add_library(airplay STATIC ${CORE})
    target_include_directories(airplay PRIVATE "${CMAKE_SOURCE_DIR}/src")
    if(BUILD_DEVELOPMENT_PACKAGE)
        target_sources(airplay PRIVATE src/direct_request_diag.c)
        target_include_directories(airplay PRIVATE "${CMAKE_SOURCE_DIR}/src")
    endif()
    target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L _REENTRANT _LARGEFILE64_SOURCE _FILE_OFFSET_BITS=64)
    target_include_directories(airplay PUBLIC "${UX}/lib" PRIVATE "${UX}/lib/llhttp" "${UX}/lib/playfair" "${DEPS}/include")
    target_link_libraries(airplay PUBLIC playfair dnssd llhttp "${DEPS}/lib/libplist-2.0.a" "${DEPS}/lib/libcrypto.a" pthread dl m)
    add_executable(uxair src/main.cpp src/receiver.cpp src/audio.cpp src/h264.cpp
        src/hevc_sample.cpp src/hevc_stream.cpp src/system_bus.cpp)
    target_include_directories(uxair PRIVATE "${DEPS}/include" ${LS2_INCLUDE_DIRS})
    target_link_directories(uxair PRIVATE ${LS2_LIBRARY_DIRS})
    target_link_libraries(uxair PRIVATE lg_media airplay ${LS2_LIBRARIES} "${DEPS}/lib/libavcodec.a" "${DEPS}/lib/libswresample.a" "${DEPS}/lib/libavutil.a")
    # LG's player loads its own older OpenSSL/FFmpeg indirectly. Keep private
    # static symbols out of the dynamic namespace to avoid interposition.
    target_link_options(uxair PRIVATE -Wl,--exclude-libs,ALL)
    set_target_properties(uxair PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
